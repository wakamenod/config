function cycleForward() {
  const focused = App.focusedApp()
  if (focused.name === 'Emacs') {
    toggleEmacs();
  } else {
    cycleForwardOtherApps();
  }

function cycleBack() {
  const focused = App.focusedApp()
  if (focused.name === 'Emacs') {
    toggleEmacs();
  } else {
    cycleBackOtherApps();
  }

function cycleForwardOtherApps() {
  idx++;
  if (idx === allApps.length) {
    allApps = allAppExceptEmacs();
    idx = 0;
  }
  focusOtherApps();
}

function cycleBackOtherApps() {
  idx--;
  if (idx < 0) {
    allApps = allAppExceptEmacs();
    idx = allApps.length - 1;
  }
  focusOtherApps();
}

function focusOtherApps() {
  allApps[idx].activate(true);
  allApps[idx].mainWindow().focus();
  focusApp(allApps[idx].name);

function emacs() {
  App.open('/usr/local/Cellar/emacs-plus@28/28.0.50/Emacs.app').then(app => {
    if (app) {
      const w = Screen.currentScreen().fullFrame().width;
      const h = Screen.currentScreen().fullFrame().height;

function vertical(isLeft) {
  // some window doesn't go where intended to for the first time so...
  for (let i = 0; i < 2; i++) {
    const wHalf = Screen.currentScreen().fullFrame().width / 2;
    const hFull = Screen.currentScreen().fullFrame().height;
    Window.allWindows().forEach(w => {
      w.setSize({ width: wHalf, height: hFull });
      if (w.app.name === 'Emacs' || w.app.name === 'IntelliJ IDEA') {
        w.setPosition({ x: (isLeft ? 0 : wHalf), y: 0 });
      } else {
        w.setPosition({ x: (isLeft ? wHalf : 0), y: 0 });
      }
    })
  }
  isEmacsFull = false;
}

Keyboard.onModsChanged = (mods) => {
    if (mods.Command && mods.Shift && mods.Fn) {
        if (!Mouse.onMoved) {
            const win = Window.windowUnderMouse();
            const offset = win.position().minus(Mouse.position());
            Mouse.onMoved = (p) => {
                win.setPosition(p.plus(offset));
            }
        }
    }
    else {
        if (Mouse.onMoved) {
            Mouse.onMoved = null;
        }
    }
};

function dance(win: Window) {
    const f = win.position();
    const w = (Math.random() * 0.6 + 0.4) * 200;
    const h = (Math.random() * 0.6 + 0.4) * 200;

            y: f.y + (Math.sin(t / h) * r),

let restorers: { [id: string]: () => void } = {};

Hotkey.activate(['command', 'option', 'control'], 'm', () => {
    let win = Window.focusedWindow();
    if (!restorers[win.id]) {
        let oldFrame = win.frame();

        Window.focusedWindow().maximize();
    }
});

    Object.values(restorers).forEach(fn => fn());
    restorers = {};
});

function hhmmss(time) {
    const s = time % 60;
    time = Math.floor(time / 60);
    const m = time % 60;
    time = Math.floor(time / 60);
    const h = time;
    return [h, m, s].map(padWithZeroes).join(':');
}

function padWithZeroes(n: number) {
    return n.toString().padStart(2, '0');
}
