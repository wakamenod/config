    (interactive "p")

(leaf *smart-kill
  :bind
  (("C-c C-M-d" . kill-word-at-point)
   ("C-w"       . backward-kill-word-or-region))
  :init
  (defun kill-word-at-point ()
    (interactive)
    (let ((char (char-to-string (char-after (point)))))
      (cond
       ((string= " " char) (delete-horizontal-space))
       ((string-match "[\t\n -@\[-`{-~],.、。" char) (kill-word 1))
       (t (forward-char) (backward-word) (kill-word 1)))))
  (defun backward-kill-word-or-region (&optional arg)
    (interactive "p")
    (if (region-active-p)
        (call-interactively #'kill-region)
      (backward-kill-word arg))))

(defun zenn-paste-image-from-clipboard (name)
  "クリップボードの画像を Git ルート直下の images/ ディレクトリに保存し、Zenn用のMarkdownリンクを挿入します。"
  (interactive "s画像名を入力 (拡張子なし): ")
  (let* ((git-root (locate-dominating-file default-directory ".git"))
         (root-dir (or git-root default-directory)) ; .git が見つからない場合は現在のディレクトリを使用
         (image-dir (expand-file-name "images/" root-dir))
         (file-path (expand-file-name (concat name ".png") image-dir))
         (markdown-path (format "/images/%s.png" name)))

    ;; imagesディレクトリが存在しない場合は作成する
    (unless (file-exists-p image-dir)
      (make-directory image-dir t))

    ;; pngpasteコマンドを実行して画像を保存
    (if (eq 0 (call-process "pngpaste" nil nil nil file-path))
        (progn
          (insert (format "![%s](%s)" name markdown-path))
          (message "画像を保存しました: %s" file-path))
      (message "エラー: クリップボードに画像がないか、pngpasteが利用できません。"))))
